\section{Profile}
\cvitem{}{Graduating from the University of W\"urzburg in April 2027, advised by Carlo D'Eramo. I started out designing algorithms for exploration and policy optimization, and now study the learning dynamics of neural networks, in particular memory and plasticity in continual learning, and how reinforcement learning changes them compared with supervised learning.}

\section{Publications}
\pub{2026}{\textbf{Mahdi Kallel}, Nour Jamoussi, Johannes T\"olle, Marios Kountouris, Carlo D'Eramo}
  {https://openreview.net/forum?id=Kgz0zlJVS6}{The Spectral Anatomy of Plasticity Loss}{AXIOM Workshop at NeurIPS 2026}
  {\summary{Uses random matrix theory to explain loss of plasticity, and shows that protecting the directions that hold old knowledge keeps networks trainable.}}

\pub{2026}{\textbf{Mahdi Kallel}, Johannes T\"olle, Ahmed Hendawy, Carlo D'Eramo}
  {https://arxiv.org/abs/2604.22110}{Do Not Imitate, Reinforce: Iterative Classification via Belief Refinement}{Transactions on Machine Learning Research (TMLR), 2026}
  {\summary{Trains classifiers by reinforcement learning, rewarding every step that improves the prediction. Achieves a better balance of accuracy and calibration than cross-entropy and stays robust to label noise.}}

\pub{2026}{\textbf{Mahdi Kallel}, Jose-Luis Holgado-Alvarez, Samuele Tosatto, Carlo D'Eramo}
  {https://rlj.cs.umass.edu/2026/papers/Paper8.html}{PPO+: Enhancing Proximal Policy Optimization}{Reinforcement Learning Journal (RLC 2026)}
  {\summary{Shows that PPO is held back by its critic, and fixes it with an off-policy critic.}}

\pub{2026}{Ahmed Hendawy, Henrik Metternich, Th\'eo Vincent, \textbf{Mahdi Kallel}, Jan Peters, Carlo D'Eramo}
